#include <stdio.h>

long long gcd(long long a, long long b) {
    return b == 0 ? a : gcd(b, a % b);
}

long long modinv(long long a, long long m) {
    long long m0 = m, y = 0, x = 1;
    while (a > 1) {
        long long q = a / m, t = m;
        m = a % m; a = t; t = y;
        y = x - q * y; x = t;
    }
    return (x + m0) % m0;
}

int main() {
    long long p = 59, q = 61, n = p * q;
    long long e_old = 31, d_old = 3031;

    // Knowing (e_old, d_old), attacker knows e_old * d_old - 1 = k * phi(n)
    long long kphi = e_old * d_old - 1;
    // Factor out powers of 2 from kphi: kphi = 2^s * r
    long long r = kphi;
    while (r % 2 == 0) r /= 2;

    // Using base a = 2 to find non-trivial square root of 1 mod n
    long long x = 1, base = 2, exp = r;
    while (exp > 0) {
        if (exp & 1) x = (x * base) % n;
        base = (base * base) % n;
        exp >>= 1;
    }
    while ((x * x) % n != 1) x = (x * x) % n;

    long long factored_p = gcd(x - 1, n);
    long long factored_q = n / factored_p;
    long long phi = (factored_p - 1) * (factored_q - 1);

    // Bob generates new key pair (e_new, d_new) with the SAME modulus n
    long long e_new = 37;
    long long d_new = modinv(e_new, phi);

    printf("No, it is NOT safe!\n");
    printf("Knowing (e_old, d_old) allows an attacker to factor n into p=%lld and q=%lld.\n", factored_p, factored_q);
    printf("With phi(n)=%lld known, any new public key e_new=%lld yields d_new=%lld immediately.\n", phi, e_new, d_new);
    return 0;
}
