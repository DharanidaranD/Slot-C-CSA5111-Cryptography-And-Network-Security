#include <stdio.h>

long long modpow(long long b, long long e, long long m) {
    long long r = 1;
    b %= m;
    while (e > 0) {
        if (e & 1) r = (r * b) % m;
        b = (b * b) % m;
        e >>= 1;
    }
    return r;
}

long long modinv(long long a, long long m) {
    long long m0 = m, y = 0, x = 1;
    while (a > 1) {
        long long q = a / m, t = m;
        m = a % m; a = t; t = y;
        y = x - q * y; x = t;
    }
    return (x + m0) % m0;
}

int main() {
    // RSA parameters (deterministic)
    long long n = 3599, d = 3031, H = 42;
    long long rsa_sig1 = modpow(H, d, n);
    long long rsa_sig2 = modpow(H, d, n);

    // Toy DSA parameters (randomized k per signature)
    long long p = 23, q = 11, g = 4, x = 3; // y = g^x mod p = 18
    long long k1 = 5, k2 = 7;

    long long r1 = modpow(g, k1, p) % q;
    long long s1 = (modinv(k1, q) * (H + x * r1)) % q;

    long long r2 = modpow(g, k2, p) % q;
    long long s2 = (modinv(k2, q) * (H + x * r2)) % q;

    printf("RSA Signature 1: %lld | RSA Signature 2: %lld (Identical)\n", rsa_sig1, rsa_sig2);
    printf("DSA Signature 1 (k=%lld): (%lld, %lld)\n", k1, r1, s1);
    printf("DSA Signature 2 (k=%lld): (%lld, %lld) (Different!)\n\n", k2, r2, s2);
    printf("Implications:\n");
    printf("1. Security: DSA prevents attackers from detecting duplicate signed messages.\n");
    printf("2. Critical Risk: If k is ever reused across two different messages in DSA,\n");
    printf("   an attacker can immediately solve for the private key x!\n");
    return 0;
}
