#include <stdio.h>

int modinv(int a, int m) {
    a = (a % m + m) % m;
    for (int x = 1; x < m; x++)
        if ((a * x) % m == 1) return x;
    return -1;
}

int main() {
    // Known plaintext matrix P (columns: "HE", "LP" -> [7, 11; 4, 15])
    int P[2][2] = {{7, 11}, {4, 15}};
    // Actual secret key K = [3, 3; 2, 5]
    int K[2][2] = {{3, 3}, {2, 5}};
    int C[2][2] = {0};

    // Compute C = K * P mod 26
    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            for (int k = 0; k < 2; k++)
                C[i][j] = (C[i][j] + K[i][k] * P[k][j]) % 26;

    // Attacker recovers K = C * P^-1 mod 26
    int det = (P[0][0] * P[1][1] - P[0][1] * P[1][0]) % 26;
    int det_inv = modinv(det, 26);

    int P_inv[2][2];
    P_inv[0][0] = ( P[1][1] * det_inv) % 26;
    P_inv[0][1] = ((-P[0][1] + 26) * det_inv) % 26;
    P_inv[1][0] = ((-P[1][0] + 26) * det_inv) % 26;
    P_inv[1][1] = ( P[0][0] * det_inv) % 26;

    int K_rec[2][2] = {0};
    for (int i = 0; i < 2; i++)
        for (int j = 0; j < 2; j++)
            for (int k = 0; k < 2; k++)
                K_rec[i][j] = (K_rec[i][j] + C[i][k] * P_inv[k][j]) % 26;

    printf("Original Key Matrix : [[%d, %d], [%d, %d]]\n", K[0][0], K[0][1], K[1][0], K[1][1]);
    printf("Recovered Key Matrix: [[%d, %d], [%d, %d]]\n", K_rec[0][0], K_rec[0][1], K_rec[1][0], K_rec[1][1]);
    return 0;
}
