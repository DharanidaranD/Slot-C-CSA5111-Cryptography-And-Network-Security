#include <stdio.h>
#include <string.h>

int main() {
    char pt[] = "sendmoremoney";
    int key[] = {9, 0, 1, 7, 23, 15, 21, 14, 11, 11, 2, 8, 9};
    int len = strlen(pt);
    char ct[20];

    printf("a. Ciphertext: ");
    for (int i = 0; i < len; i++) {
        ct[i] = ((pt[i] - 'a' + key[i]) % 26) + 'A';
        printf("%c", ct[i]);
    }
    ct[len] = '\0';
    printf("\n");

    char target[] = "cashnotneeded";
    printf("b. Key to decrypt '%s' to '%s':\n   ", ct, target);
    for (int i = 0; i < len; i++) {
        int k = ((ct[i] - 'A') - (target[i] - 'a') + 26) % 26;
        printf("%d ", k);
    }
    printf("\n");
    return 0;
}
