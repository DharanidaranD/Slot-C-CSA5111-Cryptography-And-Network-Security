#include <stdio.h>

int P10[] = {3,5,2,7,4,10,1,9,8,6}, P8[] = {6,3,7,4,8,5,10,9};
int IP[] = {2,6,3,1,4,8,5,7}, II[] = {4,1,3,5,7,2,8,6};
int EP[] = {4,1,2,3,2,3,4,1}, P4[] = {2,4,3,1};
int S0[4][4] = {{1,0,3,2},{3,2,1,0},{0,2,1,3},{3,1,3,2}};
int S1[4][4] = {{0,1,2,3},{2,0,1,3},{3,0,1,0},{2,1,0,3}};
int K1[8], K2[8];

void pm(int *a, int *b, int *p, int n) {
    for (int i = 0; i < n; i++) b[i] = a[p[i] - 1];
}

void ls(int *a) {
    int t = a[0];
    for (int i = 0; i < 4; i++) a[i] = a[i + 1];
    a[4] = t;
}

void kg(int *k) {
    int t[10], l[5], r[5];
    pm(k, t, P10, 10);
    for (int i = 0; i < 5; i++) { l[i] = t[i]; r[i] = t[i + 5]; }
    ls(l); ls(r);
    for (int i = 0; i < 5; i++) { t[i] = l[i]; t[i + 5] = r[i]; }
    pm(t, K1, P8, 8);
    ls(l); ls(l); ls(r); ls(r);
    for (int i = 0; i < 5; i++) { t[i] = l[i]; t[i + 5] = r[i]; }
    pm(t, K2, P8, 8);
}

int sb(int *x, int s[4][4]) {
    return s[x[0]*2 + x[3]][x[1]*2 + x[2]];
}

void fk(int *x, int *k) {
    int e[8], v[4], q[4];
    pm(x, e, EP, 8);
    for (int i = 0; i < 8; i++) e[i] ^= k[i];
    v[0] = sb(e, S0) >> 1; v[1] = sb(e, S0) & 1;
    v[2] = sb(e + 4, S1) >> 1; v[3] = sb(e + 4, S1) & 1;
    pm(v, q, P4, 4);
    for (int i = 0; i < 4; i++) x[i] ^= q[i];
}

void sd(int *a, int *b) {
    int t[8], z[8], q;
    pm(a, t, IP, 8);
    fk(t, K1);
    for (int i = 0; i < 4; i++) { q = t[i]; t[i] = t[i + 4]; t[i + 4] = q; }
    fk(t, K2);
    pm(t, z, II, 8);
    for (int i = 0; i < 8; i++) b[i] = z[i];
}

int val(int *b) {
    int v = 0;
    for (int i = 0; i < 8; i++) v = v * 2 + b[i];
    return v;
}

void bits(int x, int *b) {
    for (int i = 7; i >= 0; i--) b[7 - i] = (x >> i) & 1;
}

void show(int x) {
    for (int i = 7; i >= 0; i--) printf("%d", (x >> i) & 1);
}

int main() {
    int key[10] = {0,1,1,1,1,1,1,1,0,1};
    int p[3] = {1, 2, 4}, c[3], d[3], ctr[8], o[8];

    kg(key);
    for (int i = 0; i < 3; i++) {
        bits(i, ctr);
        sd(ctr, o);
        c[i] = p[i] ^ val(o);
        d[i] = c[i] ^ val(o);
    }

    printf("Ciphertext: ");
    for (int i = 0; i < 3; i++) show(c[i]);
    printf("\nDecrypted : ");
    for (int i = 0; i < 3; i++) show(d[i]);
    printf("\nExpected  : 001110000100111100110010\n");
    return 0;
}
