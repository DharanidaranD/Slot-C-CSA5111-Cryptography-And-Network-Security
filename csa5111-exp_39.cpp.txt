#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main() {
    // Standard English letter frequencies (%)
    const double eng[26] = {
        8.17, 1.49, 2.78, 4.25, 12.70, 2.23, 2.02, 6.09, 6.97, 0.15, 0.77, 4.03, 2.41,
        6.75, 7.51, 1.93, 0.10, 5.99, 6.33, 9.06, 2.76, 0.98, 2.36, 0.15, 1.97, 0.07
    };

    char ct[512];
    int top_n;
    printf("Enter additive ciphertext: ");
    fgets(ct, sizeof(ct), stdin);
    printf("Enter number of top plaintexts to display (e.g., 10): ");
    scanf("%d", &top_n);

    double score[26];
    int shift[26];

    for (int k = 0; k < 26; k++) {
        shift[k] = k;
        int cnt[26] = {0}, total = 0;
        for (int i = 0; ct[i]; i++) {
            if (isalpha(ct[i])) {
                int p = (toupper(ct[i]) - 'A' - k + 26) % 26;
                cnt[p]++;
                total++;
            }
        }
        // Chi-squared goodness-of-fit statistic (lower = closer to English)
        score[k] = 0.0;
        for (int i = 0; i < 26; i++) {
            double expected = total * (eng[i] / 100.0);
            double diff = cnt[i] - expected;
            score[k] += (diff * diff) / expected;
        }
    }

    // Sort keys by ascending chi-squared score
    for (int i = 0; i < 25; i++) {
        for (int j = i + 1; j < 26; j++) {
            if (score[j] < score[i]) {
                double ts = score[i]; score[i] = score[j]; score[j] = ts;
                int tk = shift[i]; shift[i] = shift[j]; shift[j] = tk;
            }
        }
    }

    for (int r = 0; r < top_n && r < 26; r++) {
        int k = shift[r];
        printf("#%2d (Key=%2d): ", r + 1, k);
        for (int i = 0; ct[i] && ct[i] != '\n'; i++) {
            if (isalpha(ct[i])) printf("%c", ((toupper(ct[i]) - 'A' - k + 26) % 26) + 'A');
            else printf("%c", ct[i]);
        }
        printf("\n");
    }
    return 0;
}
